// Pomocnicza funkcja czyszcząca i unifikująca niemieckie znaki function normalizeGermanStr(str) { if (!str) return ""; return str.toLowerCase() .replace(/ä/g, 'ae') .replace(/ö/g, 'oe') .replace(/ü/g, 'ue') .replace(/ß/g, 'ss') .replace(/^(landkreis|kreis|stadt)\s+/i, '') .trim(); } // 1. Poprawione filtrowanie powiatów (znajduje "gießen", "giessen", "borken", "goettingen" itp.) function filterCounties() { const rawQuery = document.getElementById('dabuRegionSearch').value; const query = normalizeGermanStr(rawQuery); const selectedLand = document.getElementById('landSelectFilter').value; const list = document.getElementById('countiesSearchList'); list.innerHTML = ''; let filtered = allCountiesGeoList.filter(c => { const normName = normalizeGermanStr(c.name); const normLand = normalizeGermanStr(c.land); const matchesQuery = normName.includes(query) || normLand.includes(query); const matchesLand = !selectedLand || c.land === selectedLand; return matchesQuery && matchesLand; }); const groupedByLand = {}; filtered.forEach(c => { if (!groupedByLand[c.land]) groupedByLand[c.land] = []; groupedByLand[c.land].push(c); }); const sortedLands = Object.keys(groupedByLand).sort(); if (sortedLands.length === 0) { list.innerHTML = `
Brak pasujących powiatów
`; return; } sortedLands.forEach(land => { const header = document.createElement('div'); header.className = 'land-group-header'; header.innerText = `${land} (${groupedByLand[land].length})`; list.appendChild(header); groupedByLand[land].sort((a,b) => a.name.localeCompare(b.name, 'de')).forEach(c => { const item = document.createElement('div'); item.className = 'county-search-item'; item.innerHTML = ` 📍 ${c.name} Wybierz › `; item.onclick = () => { list.classList.remove('is-open'); map.fitBounds(c.bounds, { padding: [60, 60] }); fetchMunicipalitiesInFlight(c.name, c.land, c.layer); }; list.appendChild(item); }); }); } // 2. Niezawodne pobieranie gmin w locie po ramce BBOX klikniętego powiatu async function fetchMunicipalitiesInFlight(countyName, landName, layer) { const modalTitle = document.getElementById('modalCountyTitle'); const container = document.getElementById('modalMunicipalitiesList'); modalTitle.innerHTML = `📍 ${countyName} (${landName})`; container.innerHTML = `
⏳

Pobieranie gmin w locie z bazy danych...

`; document.getElementById('countyDetailsModal').classList.add('is-active'); const cleanName = countyName.replace(/^(Landkreis|Kreis|Stadt)\s+/i, '').trim(); // Pobieramy współrzędne obrysu powiatu (South, West, North, East) let bbox = ""; if (layer && layer.getBounds) { const b = layer.getBounds(); bbox = `${b.getSouth()},${b.getWest()},${b.getNorth()},${b.getEast()}`; } // Zapytanie Overpass: pobiera wszystkie gminy (admin_level 8) oraz miasta/wioski wewnątrz granic powiatu let overpassQuery = ""; if (bbox) { overpassQuery = ` [out:json][timeout:25]; ( relation["boundary"="administrative"]["admin_level"="8"](${bbox}); node["place"~"^(city|town|village)$"](${bbox}); ); out tags; `; } else { overpassQuery = ` [out:json][timeout:25]; area["name"~"${cleanName}"]["boundary"="administrative"]->.a; ( relation["boundary"="administrative"]["admin_level"="8"](area.a); node["place"~"^(city|town|village)$"](area.a); ); out tags; `; } const encoded = encodeURIComponent(overpassQuery); const endpoints = [ "https://overpass-api.de/api/interpreter?data=" + encoded, "https://overpass.kumi.systems/api/interpreter?data=" + encoded ]; let municipalities = []; let success = false; for (let url of endpoints) { try { const resp = await fetch(url); if (!resp.ok) continue; const data = await resp.json(); if (data && data.elements && data.elements.length > 0) { data.elements.forEach(el => { if (el.tags && el.tags.name) { const n = el.tags.name.trim(); // Odrzucamy same nazwy powiatów i landów if (!n.toLowerCase().startsWith("landkreis") && !n.toLowerCase().startsWith("kreis ") && n !== countyName) { municipalities.push(n); } } }); success = true; break; } } catch (e) { console.warn("Błąd serwera Overpass, próba kolejnego...", e); } } municipalities.push(cleanName); municipalities = [...new Set(municipalities)].sort((a, b) => a.localeCompare(b, 'de')); if (success && municipalities.length > 0) { container.innerHTML = ''; municipalities.forEach(mun => { const row = document.createElement('div'); row.className = 'municipality-row'; const isAdded = selectedMunicipalities.includes(mun); row.innerHTML = ` ${mun} `; container.appendChild(row); }); } else { container.innerHTML = `

Nie udało się rozwinąć listy gmin dla tego obszaru.
Możesz złożyć zapytanie ogólne o cały powiat.

`; } }
